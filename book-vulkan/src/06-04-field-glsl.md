# 6.4 域运算从 Rust 到 GLSL 的对照

这节给出 carry、加减和乘法（fe 结构体与常量 6.3 节已敲）。翻译是机械的，先讲三条规则，再贴代码。

## 三条翻译规则

1. `struct Fe(pub [u64; 10])` → `struct fe { uint64_t v[10]; }`，成员从 `.0[i]` 变成 `.v[i]`；
2. Rust 的引用参数 `&Fe` → GLSL 直接按值传（结构体参数默认 `in`，返回新值），风格正好一致；
3. Rust 常量 `(1 << 26) - 1` → GLSL 十六进制字面量 `0x3ffffffUL`（记得 UL 后缀）。

## carry、add、sub

fe 结构体和 M26/M25 常量 6.3 节已经敲过；域运算从这里开始，接在 sm64_next 之后：
```glsl
// ---------------- 域运算 ----------------
fe fe_carry(fe h) {
    uint64_t c;
    c = h.v[0] >> 26; h.v[1] += c; h.v[0] &= M26;
    c = h.v[1] >> 25; h.v[2] += c; h.v[1] &= M25;
    c = h.v[2] >> 26; h.v[3] += c; h.v[2] &= M26;
    c = h.v[3] >> 25; h.v[4] += c; h.v[3] &= M25;
    c = h.v[4] >> 26; h.v[5] += c; h.v[4] &= M26;
    c = h.v[5] >> 25; h.v[6] += c; h.v[5] &= M25;
    c = h.v[6] >> 26; h.v[7] += c; h.v[6] &= M26;
    c = h.v[7] >> 25; h.v[8] += c; h.v[7] &= M25;
    c = h.v[8] >> 26; h.v[9] += c; h.v[8] &= M26;
    c = h.v[9] >> 25; h.v[0] += 19 * c; h.v[9] &= M25;
    c = h.v[0] >> 26; h.v[1] += c; h.v[0] &= M26;
    return h;
}

fe fe_add(fe a, fe b) {
    fe h;
    for (int i = 0; i < 10; i++) h.v[i] = a.v[i] + b.v[i];
    return h;
}

fe fe_sub(fe a, fe b) {
    const uint64_t two_p[10] = uint64_t[10](
        2UL * ((1UL << 26) - 19),
        2UL * ((1UL << 25) - 1), 2UL * ((1UL << 26) - 1),
        2UL * ((1UL << 25) - 1), 2UL * ((1UL << 26) - 1),
        2UL * ((1UL << 25) - 1), 2UL * ((1UL << 26) - 1),
        2UL * ((1UL << 25) - 1), 2UL * ((1UL << 26) - 1),
        2UL * ((1UL << 25) - 1)
    );
    fe h;
    for (int i = 0; i < 10; i++) h.v[i] = a.v[i] + two_p[i] - b.v[i];
    return h;
}
```

和 4.2/4.3 节逐行对照，唯一的语法差异是数组初始化写成构造函数 `uint64_t[10](...)`。

## fe_mul

和 4.4 节同一个函数，连系数变量名都一样（`f1_2`、`g9_19`……）。数组解构变成 20 行显式局部变量声明：

```glsl
fe fe_mul(fe f, fe g) {
    uint64_t f0=f.v[0], f1=f.v[1], f2=f.v[2], f3=f.v[3], f4=f.v[4];
    uint64_t f5=f.v[5], f6=f.v[6], f7=f.v[7], f8=f.v[8], f9=f.v[9];
    uint64_t g0=g.v[0], g1=g.v[1], g2=g.v[2], g3=g.v[3], g4=g.v[4];
    uint64_t g5=g.v[5], g6=g.v[6], g7=g.v[7], g8=g.v[8], g9=g.v[9];

    uint64_t f1_2 = 2*f1, f3_2 = 2*f3, f5_2 = 2*f5, f7_2 = 2*f7, f9_2 = 2*f9;
    uint64_t g1_19 = 19*g1, g2_19 = 19*g2, g3_19 = 19*g3, g4_19 = 19*g4, g5_19 = 19*g5;
    uint64_t g6_19 = 19*g6, g7_19 = 19*g7, g8_19 = 19*g8, g9_19 = 19*g9;

    fe h;
    h.v[0] = f0*g0 + f1_2*g9_19 + f2*g8_19 + f3_2*g7_19 + f4*g6_19
           + f5_2*g5_19 + f6*g4_19 + f7_2*g3_19 + f8*g2_19 + f9_2*g1_19;
    h.v[1] = f0*g1 + f1*g0 + f2*g9_19 + f3*g8_19 + f4*g7_19
           + f5*g6_19 + f6*g5_19 + f7*g4_19 + f8*g3_19 + f9*g2_19;
    h.v[2] = f0*g2 + f1_2*g1 + f2*g0 + f3_2*g9_19 + f4*g8_19
           + f5_2*g7_19 + f6*g6_19 + f7_2*g5_19 + f8*g4_19 + f9_2*g3_19;
    h.v[3] = f0*g3 + f1*g2 + f2*g1 + f3*g0 + f4*g9_19
           + f5*g8_19 + f6*g7_19 + f7*g6_19 + f8*g5_19 + f9*g4_19;
    h.v[4] = f0*g4 + f1_2*g3 + f2*g2 + f3_2*g1 + f4*g0
           + f5_2*g9_19 + f6*g8_19 + f7_2*g7_19 + f8*g6_19 + f9_2*g5_19;
    h.v[5] = f0*g5 + f1*g4 + f2*g3 + f3*g2 + f4*g1 + f5*g0
           + f6*g9_19 + f7*g8_19 + f8*g7_19 + f9*g6_19;
    h.v[6] = f0*g6 + f1_2*g5 + f2*g4 + f3_2*g3 + f4*g2 + f5_2*g1 + f6*g0
           + f7_2*g9_19 + f8*g8_19 + f9_2*g7_19;
    h.v[7] = f0*g7 + f1*g6 + f2*g5 + f3*g4 + f4*g3 + f5*g2 + f6*g1 + f7*g0
           + f8*g9_19 + f9*g8_19;
    h.v[8] = f0*g8 + f1_2*g7 + f2*g6 + f3_2*g5 + f4*g4
           + f5_2*g3 + f6*g2 + f7_2*g1 + f8*g0 + f9_2*g9_19;
    h.v[9] = f0*g9 + f1*g8 + f2*g7 + f3*g6 + f4*g5
           + f5*g4 + f6*g3 + f7*g2 + f8*g1 + f9*g0;
    return fe_carry(h);
}
```

💡 **补给：为什么没有指针也不慢。** GLSL 里结构体按值传递看起来像「每次调用复制 80 字节」，但 shader 编译器会把这些小函数全部内联，按值语义只是源码层面的写法，生成的机器码和指针版一样。放心直译。
